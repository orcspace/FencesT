cmake_minimum_required(VERSION 3.20)
project(FencesT VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(model_tests tests/model_tests.cpp)
  target_include_directories(model_tests PRIVATE src)
  if(MSVC)
    target_compile_options(model_tests PRIVATE /utf-8 /W4 /permissive-)
  endif()
  add_test(NAME state_roundtrip_and_validation COMMAND model_tests)
endif()
if(WIN32)
  enable_language(RC)
  add_executable(FencesT WIN32 src/main.cpp src/platform.cpp src/render.cpp src/panel.cpp src/settings.cpp resources/app.rc)
  target_compile_definitions(FencesT PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(FencesT PRIVATE gdiplus comctl32 comdlg32 shell32 shlwapi ole32 uuid dwmapi user32 gdi32 advapi32)
  if(MSVC)
    target_compile_options(FencesT PRIVATE /W4 /permissive- /utf-8 /EHsc)
    set_property(TARGET FencesT PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(FencesT PRIVATE /MANIFEST:NO)
  else()
    target_compile_options(FencesT PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(FencesT PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  install(TARGETS FencesT RUNTIME DESTINATION .)
  install(FILES README.md docs/QUICKSTART.ru.md DESTINATION .)
  set(CPACK_GENERATOR ZIP)
  set(CPACK_PACKAGE_FILE_NAME "FencesT-${PROJECT_VERSION}-windows-x64")
  include(CPack)
endif()
